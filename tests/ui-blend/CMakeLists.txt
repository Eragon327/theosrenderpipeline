cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(TRPUIBlendTests LANGUAGES CXX)
    find_package(spdlog CONFIG REQUIRED)
    enable_testing()
endif()
add_executable(UIBlendTests UIBlendTests.cpp ../../src/FrameGen/NativeUIComposition.cpp)
target_compile_features(UIBlendTests PRIVATE cxx_std_20)
target_compile_definitions(UIBlendTests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(UIBlendTests PRIVATE ../../src)
target_link_libraries(UIBlendTests PRIVATE spdlog::spdlog d3d11 d3dcompiler dxgi)
add_test(NAME NativeUIComposition COMMAND UIBlendTests)
add_executable(UIBlendCoverageTests UIBlendCoverageTests.cpp ../../src/FrameGen/NativeUIComposition.cpp)
target_compile_features(UIBlendCoverageTests PRIVATE cxx_std_20)
target_compile_definitions(UIBlendCoverageTests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(UIBlendCoverageTests PRIVATE ../../src)
target_link_libraries(UIBlendCoverageTests PRIVATE spdlog::spdlog d3d11 d3dcompiler dxgi)
add_test(NAME NativeUIBlendCoverage COMMAND UIBlendCoverageTests)
set_tests_properties(NativeUIComposition NativeUIBlendCoverage PROPERTIES TIMEOUT 30)
add_executable(UIHookContinuityTests UIHookContinuityTests.cpp ../../src/FrameGen/NativeUIComposition.cpp)
target_compile_features(UIHookContinuityTests PRIVATE cxx_std_20)
target_compile_definitions(UIHookContinuityTests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(UIHookContinuityTests PRIVATE ../../src)
target_link_libraries(UIHookContinuityTests PRIVATE spdlog::spdlog d3d11 d3dcompiler dxgi)
add_test(NAME NativeUIHookContinuity COMMAND UIHookContinuityTests)
set_tests_properties(NativeUIHookContinuity PROPERTIES TIMEOUT 30)
